                <div class="bg-white rounded-2xl p-6 shadow-sm border border-slate-200 hover:shadow-md transition">
                    <div class="flex items-center justify-between border-b pb-3 mb-4">
                        <h3 class="font-bold text-lg text-slate-800"><i class="fa-solid fa-arrow-right-smooth text-blue-500 mr-2"></i>ลำดับและอนุกรมเลขคณิต</h3>
                        <span class="bg-blue-100 text-blue-700 text-xs px-2.5 py-1 rounded-full font-semibold">Arithmetic</span>
                    </div>
                    <div class="space-y-4">
                        <div class="bg-slate-50 p-4 rounded-xl border border-slate-100">
                            <p class="text-xs text-slate-500 mb-1 font-semibold">พจน์ที่ n ($a_n$) ของลำดับเลขคณิต:</p>
                            <p class="text-lg text-blue-700 font-mono">$a_n = a_1 + (n - 1)d$</p>
                            <p class="text-xs text-slate-500 mt-1">* $a_1$ คือพจน์แรก, $d$ คือผลต่างร่วม ($a_n - a_{n-1}$)</p>
                        </div>
                        <div class="bg-slate-50 p-4 rounded-xl border border-slate-100">
                            <p class="text-xs text-slate-500 mb-1 font-semibold">ผลบวก n พจน์แรก ($S_n$) ของอนุกรมเลขคณิต:</p>
                            <p class="text-lg text-blue-700 font-mono">$S_n = \frac{n}{2}(a_1 + a_n) = \frac{n}{2}[2a_1 + (n - 1)d]$</p>
                        </div>
                    </div>
                </div>

                <!-- Geometric Sequence & Series -->
                <div class="bg-white rounded-2xl p-6 shadow-sm border border-slate-200 hover:shadow-md transition">
                    <div class="flex items-center justify-between border-b pb-3 mb-4">
                        <h3 class="font-bold text-lg text-slate-800"><i class="fa-solid fa-chart-line-up text-indigo-500 mr-2"></i>ลำดับและอนุกรมเรขาคณิต</h3>
                        <span class="bg-indigo-100 text-indigo-700 text-xs px-2.5 py-1 rounded-full font-semibold">Geometric</span>
                    </div>
                    <div class="space-y-4">
                        <div class="bg-slate-50 p-4 rounded-xl border border-slate-100">
                            <p class="text-xs text-slate-500 mb-1 font-semibold">พจน์ที่ n ($a_n$) ของลำดับเรขาคณิต:</p>
                            <p class="text-lg text-indigo-700 font-mono">$a_n = a_1 \cdot r^{n-1}$</p>
                            <p class="text-xs text-slate-500 mt-1">* $a_1$ คือพจน์แรก, $r$ คืออัตราส่วนร่วม ($a_n / a_{n-1}$)</p>
                        </div>
                        <div class="bg-slate-50 p-4 rounded-xl border border-slate-100">
                            <p class="text-xs text-slate-500 mb-1 font-semibold">ผลบวก n พจน์แรก ($S_n$) และอนุกรมอนันต์ ($S_\infty$):</p>
                            <p class="text-lg text-indigo-700 font-mono">$S_n = \frac{a_1(1 - r^n)}{1 - r} \quad (r \neq 1)$</p>
                            <p class="text-md text-emerald-600 font-mono mt-1">$S_\infty = \frac{a_1}{1 - r} \quad (|r| < 1)$</p>
                        </div>
                    </div>

        function updateGraph() {
            if (!chartInstance) return;
            const type = document.getElementById('graphType').value;
            const a1 = parseFloat(document.getElementById('inputA1').value);
            const param = parseFloat(document.getElementById('inputParam').value);

            document.getElementById('valA1').innerText = a1;
            document.getElementById('valParam').innerText = param;
            document.getElementById('paramLabel').innerHTML = (type === 'arithmetic') ? 
                `ผลต่างร่วม ($d$): <span class="text-indigo-600 font-bold">${param}</span>` : 
                `อัตราส่วนร่วม ($r$): <span class="text-indigo-600 font-bold">${param}</span>`;

            let dataPoints = [];
            for (let n = 1; n <= 8; n++) {
                if (type === 'arithmetic') {
                    dataPoints.push(a1 + (n - 1) * param);
                } else {
                    dataPoints.push(a1 * Math.pow(param, n - 1));
                }
            }

            chartInstance.data.datasets[0].data = dataPoints;
            chartInstance.update();
            if (window.MathJax) MathJax.typesetPromise();
        }

        async function dbFetch(path, method = 'GET', data) {
            const url = databaseUrl(path);
            if(!url) return null;

                method,
                headers: method === 'GET' ? {} : {'Content-Type': 'application/json'},
                body: data === undefined ? undefined : JSON.stringify(data)
            });
            if(!response.ok) throw Error('ฐานข้อมูลตอบกลับ ' + response.status);
            return response.json();

        async function loadHandModel() {
          if(handLandmarker) return handLandmarker;
          let failure;
          for(const base of ['https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.22','https://unpkg.com/@mediapipe/tasks-vision@0.10.22']){
            try{
              const {HandLandmarker, FilesetResolver} = await import(base + '/+esm');
              const vision = await FilesetResolver.forVisionTasks(base + '/wasm');
              handLandmarker = await HandLandmarker.createFromOptions(vision, {
                baseOptions: {
                  modelAssetPath: 'https://storage.googleapis.com/mediapipe-models/hand_landmarker/hand_landmarker/float16/1/hand_landmarker.task',
                  delegate: 'CPU'
                },
                runningMode: 'VIDEO',
                numHands: 1,
                minHandDetectionConfidence: .55,
                minTrackingConfidence: .5
              });
              return handLandmarker;
            }catch(e){failure = e; console.warn('MediaPipe source unavailable:', base, e);}
          }
          throw failure || new Error('ไม่สามารถโหลดโมเดลตรวจจับมือ');
        }

        function describeCameraError(err) {
          const name = err?.name || '', msg = err?.message || String(err);
          if(name === 'NotAllowedError' || name === 'PermissionDeniedError' || name === 'SecurityError') 
            return 'เบราว์เซอร์ปฏิเสธสิทธิ์กล้อง: คลิกไอคอนแม่กุญแจ/กล้องบริเวณแถบที่อยู่ด้านบน → เลือก อนุญาต (Allow) การใช้กล้อง แล้วรีเฟรชหน้าเว็บ';
          if(name === 'NotFoundError' || name === 'DevicesNotFoundError') 
            return 'ไม่พบอุปกรณ์กล้องเว็บบ์แคมที่เชื่อมต่อ กรุณาตรวจสอบสายหรือฮาร์ดแวร์กล้อง';
          if(name === 'NotReadableError' || name === 'TrackStartError') 
            return 'กล้องกำลังถูกใช้งานอยู่โดยแอปพลิเคชันอื่น (เช่น Zoom, Teams) ให้ปิดโปรแกรมเหล่านั้นก่อน';
          return msg;
        }

        async function startCamera() {
          if(cameraStarting) return;
          if(cameraStream && handLandmarker){
            cameraMessage('🟢 กล้องและ MediaPipe พร้อมใช้งาน');
            if(!cameraFrame) cameraFrame = requestAnimationFrame(detectHands);
            return;
          }
          if(!window.isSecureContext || !navigator.mediaDevices?.getUserMedia){
            cameraMessage('🔴 กล้องถูกปิดกั้นเนื่องจาก URL ไม่ปลอดภัย', 'ต้องเปิดใช้งานผ่าน HTTPS หรือ http://localhost เท่านั้น');
            return;
          }
          cameraStarting = true; 
          const requestId = ++cameraRequestId;
          try {
            cameraMessage('⏳ กำลังขอสิทธิ์เปิดกล้อง...');
            if(!cameraStream){
              try {
                cameraStream = await navigator.mediaDevices.getUserMedia({audio: false, video: {facingMode: 'user', width: {ideal: 640}, height: {ideal: 480}}});
              } catch(err) {
                if(err.name === 'OverconstrainedError' || err.name === 'NotFoundError') {
                  cameraStream = await navigator.mediaDevices.getUserMedia({audio: false, video: true});
                } else {
                  throw err;
                }
              }
            }
            if(requestId !== cameraRequestId){ cameraStream?.getTracks().forEach(t => t.stop()); cameraStream = null; return; }
            
            const vid = $id('handVideo');
            vid.srcObject = cameraStream;
            $id('cameraPanel').classList.remove('hidden');
            await Promise.race([
              vid.play(),
              new Promise((_, reject) => setTimeout(() => reject(new Error('กล้องไม่ส่งภาพภายในเวลาที่กำหนด')), 12000))
            ]);

            $id('modeCamera').classList.add('active');$id('modePointer').classList.remove('active');
            cameraMessage('✅ กล้องเปิดแล้ว • กำลังโหลด MediaPipe AI...');

            try {
              await loadHandModel();
            } catch(modelErr) {
              console.error('Hand model failed:', modelErr);
              cameraMessage('🟠 กล้องเปิดได้ แต่โหลด AI ไม่สำเร็จ', 'ตรวจสอบอินเทอร์เน็ตว่าสามารถเข้าถึง CDN ได้ แล้วกดเปิดใหม่อีกครั้ง');
              return;
            }

            if(requestId !== cameraRequestId){ stopCamera(false); return; }
            cameraMessage('🟢 MediaPipe พร้อม • ชี้นิ้วและบีบนิ้วเพื่อเลือก[cite: 3]');
            cancelAnimationFrame(cameraFrame);
            cameraFrame = requestAnimationFrame(detectHands);
          } catch(err) {
            console.error('Camera start failed:', err);
            const reason = describeCameraError(err);
            stopCamera(false);
            cameraMessage('🔴 เปิดกล้องไม่สำเร็จ', reason);
          } finally {
            cameraStarting = false;
          }
        }

        function stopCamera(showMessage = true) {
          ++cameraRequestId;
          cancelAnimationFrame(cameraFrame);
          cameraFrame = 0;
          if(cameraStream) {
            cameraStream.getTracks().forEach(t => t.stop());
            cameraStream = null;

          $id('cameraPanel').classList.add('hidden');$id('gestureCursor').style.display = 'none';

          lastPinch = false;
          hoverElement = null;
        }

        function detectHands(timestamp) {
          if(!cameraStream || !handLandmarker) return;
          cameraFrame = requestAnimationFrame(detectHands);
          if(timestamp - lastDetection < 55) return;
          lastDetection = timestamp;
          const video = $id('handVideo');
          if(video.readyState < 2 || !video.videoWidth) return;
          try {
            const result = handLandmarker.detectForVideo(video, timestamp);
            const hand = result.landmarks?.[0], cursor = $id('gestureCursor');
            if(!hand) {
                cursor.style.display = 'none';
                $id('gestureStatus').textContent = 'ยังไม่พบมือ';
                lastPinch = false;
                return;
            }
            const point = hand[8], thumb = hand[4];
            const x = (1 - point.x) * innerWidth, y = point.y * innerHeight;
            const pinch = Math.hypot(point.x - thumb.x, point.y - thumb.y) < .055;
            cursor.style.display = 'block';
            cursor.style.left = x + 'px';
            cursor.style.top = y + 'px';
            cursor.classList.toggle('pinching', pinch);
            $id('gestureStatus').textContent = pinch ? '🤏 กำลังบีบนิ้ว' : '☝️ ชี้นิ้วไปยังตัวเลือก[cite: 3]';
            
            const el = document.elementFromPoint(x, y);
            if(hoverElement && hoverElement !== el) hoverElement.classList.remove('gesture-hover');
            hoverElement = el?.closest('button,input[type=range]');
            if(hoverElement) hoverElement.classList.add('gesture-hover');

            if(pinch && cameraScope === 'graph') {
              const slider = el?.closest('input[type=range]');
              if(slider) {
                const r = slider.getBoundingClientRect(), min = +slider.min, max = +slider.max, step = +(slider.step || 1);
                const value = Math.max(min, Math.min(max, min + Math.round(((x - r.left) / r.width * (max - min)) / step) * step));
                slider.value = value;
                slider.dispatchEvent(new Event('input', {bubbles: true}));
              }
            } else if(pinch && !lastPinch) {
              const btn = el?.closest('button');
              if(btn && ($id('sec-gesture').contains(btn) \vert{}\vert{}$id('sec-graph').contains(btn)) && !['modeCamera','modePointer'].includes(btn.id)) {
                btn.click();
              }
            }
            lastPinch = pinch;
          } catch(e) {
            console.warn('MediaPipe frame:', e);
          }
        }
